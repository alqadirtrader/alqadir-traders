// Date Normalization Helper Function (Converts any format to DD-MM-YYYY) function normalizeDate(input) { if (!input) return ''; let str = input.toString().trim().replace(/[/.]/g, '-'); let parts = str.split('-'); if (parts.length !== 3) return input; let year, month, day; if (parts[0].length === 4) { // Format: YYYY-MM-DD year = parts[0]; month = parts[1].padStart(2, '0'); day = parts[2].padStart(2, '0'); } else { // Format: D-M-YY or DD-MM-YYYY day = parts[0].padStart(2, '0'); month = parts[1].padStart(2, '0'); year = parts[2]; if (year.length === 2) { year = '20' + year; } } return `${day}-${month}-${year}`; }
